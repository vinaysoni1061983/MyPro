
1. Grade Checker
Take a score as input and print the grade based on the following:
90+ : "A"
80-89 : "B"
70-79 : "C"
60-69 : "D"
Below 60 : "F"
here we used a basic if else statement to carry out marks and all.

            #Take a score as input and print the grade based on the following:

            std_score = int(input("Enter your score : "))

            if std_score >= 60 and std_score <=69:
                print("As per the scores you got, you got grade D !")
            elif std_score >=70 and std_score <=79:
                print("As per the scores you got, you got grade C !")
            elif std_score >=80 and std_score <=89:
                print("As per the scores you got, you got grade B !")
            elif std_score >=90:
                print("As per the scores you got, you got grade A !")
            else:
                print("As per the scores you got, you got grade F !")

#We used open in read mode and file.read to read and print to display.

            fileName=input("Enter a file name to perform actions : ")

                        def getFileName():   
                            return fileName
                        
                        def openFileReadOnly():
                            openFileR=open(getFileName(),'r')
                            return openFileR
                        
                        def openFileWriteOnly():
                            openFileW=open(getFileName(),'w')
                            return openFileW
                        
                        def closeFile(actionType):
                            actionType.close()
                            
                        
                        
                        print("We are performing read only action on file : " ,getFileName())
                        myFileData=openFileReadOnly()
                        readMyData=myFileData.read()
                        print("Here are contents of the file ", getFileName(), readMyData)
                        closeFile(myFileData)
                        
                        
                        print("We are performing write only action on file : " ,getFileName())
                        myFileData=openFileWriteOnly()
                        myFileData.write("writing new data using functions to this file")
                        closeFile(myFileData)
                        
                        
                        myFileData=openFileReadOnly()
                        readMyData=myFileData.read()
                        print("Here are contents of the file ", getFileName(), "\n", readMyData)
                        closeFile(myFileData)
